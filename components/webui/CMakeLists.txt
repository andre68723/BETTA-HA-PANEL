# The www/ sources stay modular and directly serveable during development. At build time,
# bundle_assets.py combines the classic scripts into one ordered bundle, the stylesheets into one
# stylesheet in the same cascade order, and rewrites a copy of index.html with fingerprinted JS/CSS
# URLs. Only generated assets are gzip-compressed and embedded. To add a script or stylesheet, put
# it in the ordered list below and in www/index.html.

set(WEBUI_JS_DEPENDS "")
set(_webui_js_fingerprint "")
foreach(script ${WEBUI_JS_SOURCES})
    set(_source "${CMAKE_CURRENT_LIST_DIR}/www/${script}")
    list(APPEND WEBUI_JS_DEPENDS "${_source}")
    file(SHA256 "${_source}" _source_hash)
    string(APPEND _webui_js_fingerprint "${script}:${_source_hash}\n")
endforeach()
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/bundle_assets.py" _bundle_script_hash)
string(APPEND _webui_js_fingerprint "bundle_assets.py:${_bundle_script_hash}\n")
string(SHA256 _webui_js_hash "${_webui_js_fingerprint}")
string(SUBSTRING "${_webui_js_hash}" 0 16 WEBUI_JS_HASH)
set(WEBUI_CSS_DEPENDS "")
set(_webui_css_fingerprint "")
foreach(sheet ${WEBUI_CSS_SOURCES})
    set(_source "${CMAKE_CURRENT_LIST_DIR}/www/${sheet}")
    list(APPEND WEBUI_CSS_DEPENDS "${_source}")
    file(SHA256 "${_source}" _source_hash)
    string(APPEND _webui_css_fingerprint "${sheet}:${_source_hash}\n")
endforeach()
string(APPEND _webui_css_fingerprint "bundle_assets.py:${_bundle_script_hash}\n")
string(SHA256 _webui_css_hash "${_webui_css_fingerprint}")
string(SUBSTRING "${_webui_css_hash}" 0 16 WEBUI_CSS_HASH)

# Reconfigure when an input changes so the served URLs always track the asset contents.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${WEBUI_JS_DEPENDS}
    ${WEBUI_CSS_DEPENDS}
    "${CMAKE_CURRENT_LIST_DIR}/bundle_assets.py")

set(WEBUI_JS_NAME "js/app.${WEBUI_JS_HASH}.js")
set(WEBUI_CSS_NAME "styles.${WEBUI_CSS_HASH}.css")
set(WEBUI_GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/www_generated")
set(WEBUI_GEN_INDEX "${WEBUI_GEN_DIR}/index.html")
set(WEBUI_GEN_JS "${WEBUI_GEN_DIR}/${WEBUI_JS_NAME}")
set(WEBUI_GEN_CSS "${WEBUI_GEN_DIR}/${WEBUI_CSS_NAME}")

set(WEBUI_ASSETS index.html "${WEBUI_CSS_NAME}" "${WEBUI_JS_NAME}")
list(APPEND WEBUI_ASSETS "img/doorbell-example.png")

set(_assets_decl "")
set(_assets_table "")
set(_assets_gz "")
set(WEBUI_GZ_DIR "${CMAKE_CURRENT_BINARY_DIR}/www_gz")
foreach(asset ${WEBUI_ASSETS})
    get_filename_component(_base "${asset}" NAME)
    get_filename_component(_ext "${asset}" LAST_EXT)
    # The embedded symbol is _binary_<file name with every non-alphanumeric character as _>_start/_end.
    string(MAKE_C_IDENTIFIER "${_base}.gz" _id)
    if(_ext STREQUAL ".html")
        set(_mime "text/html")
    elseif(_ext STREQUAL ".css")
        set(_mime "text/css")
    elseif(_ext STREQUAL ".js")
        set(_mime "application/javascript")
    elseif(_ext STREQUAL ".png")
        set(_mime "image/png")
    else()
        message(FATAL_ERROR "webui: no content type for ${asset}")
    endif()
    string(APPEND _assets_decl
        "extern const uint8_t _binary_${_id}_start[] asm(\"_binary_${_id}_start\");\n"
        "extern const uint8_t _binary_${_id}_end[] asm(\"_binary_${_id}_end\");\n")
    string(APPEND _assets_table
        "    {\"/${asset}\", \"${_mime}\", _binary_${_id}_start, _binary_${_id}_end},\n")
    list(APPEND _assets_gz "${WEBUI_GZ_DIR}/${asset}.gz")
endforeach()
list(LENGTH WEBUI_ASSETS _asset_count)

# Write the table only when it changed, so an unchanged list does not rebuild anything.
set(_assets_c "${CMAKE_CURRENT_BINARY_DIR}/webui_assets.c")
set(_assets_content "/* Generated by components/webui/CMakeLists.txt - do not edit. */
#include \"webui_assets.h\"

${_assets_decl}
const webui_asset_t webui_assets[] = {
${_assets_table}};
const size_t webui_asset_count = ${_asset_count};
")
set(_assets_old "")
if(EXISTS "${_assets_c}")
    file(READ "${_assets_c}" _assets_old)
endif()
if(NOT _assets_old STREQUAL _assets_content)
    file(WRITE "${_assets_c}" "${_assets_content}")
endif()

idf_component_register(
    SRCS "webui_dummy.c" "${_assets_c}"
    INCLUDE_DIRS "."
)

# These files are build artifacts, never checked-in source files. One bundle reduces a cold
# editor load from 16 HTTP assets to three (HTML, CSS and JS). Register custom commands only after
# idf_component_register: ESP-IDF's early requirement scan includes this file in script mode.
idf_build_get_property(python PYTHON)
add_custom_command(
    OUTPUT "${WEBUI_GEN_INDEX}" "${WEBUI_GEN_JS}" "${WEBUI_GEN_CSS}"
    COMMAND "${python}" "${CMAKE_CURRENT_LIST_DIR}/bundle_assets.py"
        "${CMAKE_CURRENT_LIST_DIR}/www" "${WEBUI_GEN_DIR}" "${WEBUI_JS_NAME}" "${WEBUI_CSS_NAME}"
        ${WEBUI_CSS_SOURCES} ${WEBUI_JS_SOURCES}
    DEPENDS "${CMAKE_CURRENT_LIST_DIR}/bundle_assets.py"
        "${CMAKE_CURRENT_LIST_DIR}/www/index.html"
        ${WEBUI_CSS_DEPENDS} ${WEBUI_JS_DEPENDS}
    VERBATIM)

foreach(asset ${WEBUI_ASSETS})
    set(asset_src "${WEBUI_GEN_DIR}/${asset}")
    if(asset STREQUAL "img/doorbell-example.png")
        add_custom_command(
            OUTPUT "${asset_src}"
            COMMAND "${CMAKE_COMMAND}" -E make_directory "${WEBUI_GEN_DIR}/img"
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${CMAKE_CURRENT_LIST_DIR}/www/${asset}" "${asset_src}"
            DEPENDS "${CMAKE_CURRENT_LIST_DIR}/www/${asset}"
            VERBATIM)
    endif()
    set(asset_gz "${WEBUI_GZ_DIR}/${asset}.gz")
    add_custom_command(
        OUTPUT "${asset_gz}"
        COMMAND "${python}" "${CMAKE_CURRENT_LIST_DIR}/gzip_asset.py" "${asset_src}" "${asset_gz}"
        MAIN_DEPENDENCY "${asset_src}"
        DEPENDS "${CMAKE_CURRENT_LIST_DIR}/gzip_asset.py"
        VERBATIM)
    target_add_binary_data(${COMPONENT_LIB} "${asset_gz}" BINARY)
endforeach()
